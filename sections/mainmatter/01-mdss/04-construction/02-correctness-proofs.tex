We begin by defining some notational shorthand.
For a finite field $\FF$, integers $\mdssPrivThr$, and $\mdssSecret \in \FF$, define $\mdssValidPolys{\FF}{\mdssPrivThr}{\mdssSecret}$ as the set of all polynomials $p \in \FF[x]$ of degree $\mdssPrivThr$ where $p(0) = \mdssSecret$.

Next, for a field $\FF$ define the following distribution on $\FF[x]^* \times (\FF \times \FF)^*$.
\procedureblock{$\mdssHonestDist{\FF}{\mdssPrivThr}{\set{(\mdssSecret_i, \mdssNumShares_i)}_{i \in [\mdssNumShareSets]}}$}{
	\forall i \in [\mdssNumShareSets] \pcdo \\
	\t p_i \sample \mdssValidPolys{\FF}{\mdssPrivThr}{\mdssSecret_i} \\
	\mdssPolySet := \set{p_i}_{i \in [\mdssNumShareSets]} \\
	\forall i \in [\mdssNumShareSets], \forall j \in [\mdssNumShares_i] \pcdo \\
	\t x_{i, j} \sample \FF \\
	\t y_{i, j} := p_i(x_{i, j}) \\
	\pcreturn (\mdssPolySet, \set{(x_{i, j}, y_{i, j})}_{i \in [\mdssNumShareSets], j \in [\mdssNumShares_i]})
}


\newcommand{\mdssFieldBound}{\ensuremath{\left( \frac{1}{|\FF|} \right)^{\mdssRecThr - (\mdssPrivThr + 1)}}}


While $\gsDecode$ will output all polynomials of degree at most $\mdssPrivThr$ that agree on at least $\mdssRecThr$ points, that is not actually what we want.
To achieve correctness, we need $\mdssReconstructAlg$ to output only those secrets \emph{shared by the sufficient dealers}.
In other words, it should \emph{not} be the case that there are $\mdssRecThr$ points that lie on some new, separate polynomial.
Therefore, to begin our proof, we must bound away this possibility.

For a set of polynomials $\mdssPolySet$, a set of points $\mdssPointSet = \set{(x_i, y_i)}$, and integers $\mdssPrivThr$, $\mdssRecThr$, define the event $\bad_{\mdssPolySet, \mdssPointSet, \mdssPrivThr, \mdssRecThr}$ as occurring when there exists a polynomial $p^* \notin \mdssPolySet$ such that $\mathsf{deg}(p^*) \leq \mdssPrivThr$ and $\exists T \subseteq \mdssPointSet$ such that $|T| \geq \mdssRecThr$ and $\forall (x, y) \in T$, $p^*(x) = y$.
We may ocasionally drop subscripts from $\bad$ when their values are clear from context.

\begin{lemma}
	\label{lem:mdss-bad-with-polys}
	Let $\FF$ be a finite field, $\mdssPrivThr$ and $\mdssRecThr$ be positive integers such that $\mdssPrivThr < \mdssRecThr$, and P be a set of polynomials over $\FF$. Then,
	\begin{align*}
		     & \probsublong{\forall i \in [\mdssRecThr], (x_i, y_i) \sample \FF^2}{\bad_{P, \set{(x_i, y_i)}_{i \in [\mdssRecThr]}, \mdssPrivThr, \mdssRecThr}} \\
		\leq & \probsublong{\forall i \in [\mdssRecThr], (x_i, y_i) \sample \FF^2}{\bad_{P, \emptyset, \mdssPrivThr, \mdssRecThr}}
	\end{align*}
\end{lemma}
\begin{proof}
	For any set of points, if $\bad_{\mdssPolySet}$ occurs then $\bad_{\emptyset}$ must clearly also occur.
\end{proof}

\begin{claim}
	\label{cla:mdss-bad-basic-comb}
	Let $\FF$ be a finite field, and $\mdssPrivThr$, $\mdssRecThr$ be positive integers such that $\mdssPrivThr < \mdssRecThr$. Then,
	$$
		\probsublong{\forall i \in [\mdssRecThr], (x_i, y_i) \sample \FF^2}{\bad_{\emptyset, \set{(x_i, y_i)}_{i \in [\mdssRecThr]}, \mdssPrivThr, \mdssRecThr}}
		\leq \mdssFieldBound
	$$
\end{claim}
\begin{proof}
	For a set of $\mdssRecThr$ points the first $\mdssPrivThr + 1$ will uniquely determine a degree-$\mdssPrivThr$ polynomial.
	Then, the remaining $\mdssRecThr - (\mdssPrivThr + 1)$ points must all agree with that fixed polynomial, and each agrees with probability $\frac{1}{|\FF|}$.
\end{proof}

\begin{claim}
	\label{cla:mdss-bad-from-dist}
	Let $\FF$ be a finite field, $\mdssPrivThr$ and $\mdssRecThr$ be positive integers such that $\mdssPrivThr < \mdssRecThr$, and let  $S = \set{(\mdssSecret_i, \mdssNumShares_i)}_{i \in [\mdssNumShareSets]}$ where $\mdssMaxShares = \sum_{i \in [\mdssNumShareSets]} \mdssNumShares_i$.
	Then, for all $\indexSet \subseteq [\mdssMaxShares]$ where $|\indexSet| = \mdssRecThr$,
	$$
		\probsublong{(\mdssPolySet, \mdssPointSet) \sample \mdssHonestDist{\FF}{\mdssPrivThr}{S}}{\bad_{\mdssPolySet, \mdssPointSet_{\indexSet}, \mdssPrivThr, \mdssRecThr}} \leq \mdssFieldBound
	$$
	Where $\mdssPointSet_\indexSet$ indicates the set of elements of $\mdssPointSet$ as selected by the indices in $\indexSet$.
\end{claim}
\begin{proof}
	For a set of $\mdssRecThr$ points $\mdssPointSet_\indexSet$ there are two cases: either $\mdssPointSet_\indexSet$ contains $\mdssPrivThr + 1$ points on one of the polynomials $p \in \mdssPolySet$ or it does not.
	In the first case, clearly $\prob{\bad} = 0$.
	In the second, all the points in $\mdssPointSet_\indexSet$ are uninformly distributed over $\FF^2$, and so $\prob{\bad} \leq \mdssFieldBound$ by \cref{cla:mdss-bad-basic-comb} and \cref{lem:mdss-bad-with-polys}.
\end{proof}

\begin{lemma}
	\label{lem:mdss-corr-bound}
	Let $\FF$ be a finite field, $\mdssPrivThr$ and $\mdssRecThr$ be positive integers such that $\mdssPrivThr < \mdssRecThr$, and let  $S = \set{(\mdssSecret_i, \mdssNumShares_i)}_{i \in [\mdssNumShareSets]}$.
	Then,
	$$
		\probsublong{(\mdssPolySet, \mdssPointSet) \sample \mdssHonestDist{\FF}{\mdssPrivThr}{S}}{\bad_{\mdssPolySet, \mdssPointSet, \mdssPrivThr, \mdssRecThr}} \leq \binom{|\mdssPointSet|}{\mdssRecThr} \cdot \mdssFieldBound
	$$
\end{lemma}
\begin{proof}
	Note that for $\bad$ to hold there must be a size-$\mdssRecThr$ subset of $\mdssPointSet$ that agrees with a polynomial $p^* \notin \mdssPolySet$.
	By \cref{cla:mdss-bad-from-dist} we have that the probability that $\bad$ is satisfied for any size-$\mdssRecThr$ subset of $\mdssPointSet$ is at most $\mdssFieldBound$.
	The claim then follows from the union bound.
\end{proof}

Next, we must bound the probability that $\mdssShareAlg$ outputs a duplicate point, which could prevent the Guruswami-Sudan decoder from correctly recovering the secret.
For $(\mdssPolySet, \mdssPointSet) \gets \mdssHonestDist{\FF}{\mdssPrivThr}{\set{(\mdssSecret_i, \mdssNumShares_i)}_{i \in [\mdssNumShareSets]}}$, let $\bad'$ denote the event that there exists a polynomial $p_i \in \mdssPolySet$ such that $\mdssNumShares_i \geq \mdssRecThr$ and there do \emph{not} exist at least $\mdssRecThr$ distinct points $(x_j, y_j)$ in $\mdssPointSet$ such that $p_i(x_j) = y_j$.
Additionally, let $\mdssMaxShares = \sum_i \mdssNumShares_i$.
\begin{theorem}
	$$
		\prob{\bad'} \leq \left\lceil\frac{\mdssMaxShares}{\mdssRecThr}\right\rceil \cdot \frac{\mdssRecThr^2}{2|\FF|}
	$$
\end{theorem}
\begin{proof}
	Note that $\bad'$ can only occur if for a given $i$ where $\mdssNumShares_i \geq \mdssRecThr$, there is at least one $j$ such that $x_{i, j} \in \set{x_{i, 1},\dots,x_{i, j-1}}$.
\end{proof}


Our final step before proving our main theorem is show that the failure bound from the previous proof is indeed negligible.

\begin{lemma}
	\label{lem:mdss-asym-bound}
	Let $\epsilon \in (0, 1]$ be a constant, and $x, y, z, w$ be positive integer-valued polynomials in $\secpar$, where $z + 1\leq (1 - \epsilon)y$ for all values of $\secpar$.
	Then for all $w = \secpar^{\omega(1)}$ there exists a negligible function $\negl[]$ such that,
	$$
		\binom{x}{y} \cdot \left(\frac{1}{w}\right)^{y - (z + 1)} \leq \negl
	$$
\end{lemma}

\begin{proof}
	Fix a constant $c$.
	Let $\alpha$ be the constant such that $x(\secpar) \leq \secpar^\alpha$ for all sufficiently large $\secpar$.
	Note that $\frac{\alpha + c}{\epsilon}$ is also a constant.
	Let $\secpar_0$ be such that for all $\secpar \geq \secpar_0$ it holds that $x(\secpar) \leq \secpar^\alpha$ and $w(\secpar) \geq \secpar^{\frac{\alpha + c}{\epsilon}}$.
	Then, we have that
	\begin{align*}
		\binom{x}{y} \cdot \left(\frac{1}{w}\right)^{y - (z + 1)}
		 & \leq  x^y\cdot \left(\frac{1}{w}\right)^{y - (z + 1)}                                           \\
		 & \leq  \secpar^{\alpha y}\cdot \left(\frac{1}{w}\right)^{y - (z + 1)}                            \\
		 & \leq  \secpar^{\alpha y} \cdot \left(\frac{1}{w}\right)^{y - ((1 - \epsilon)y -1 + 1)}          \\
		 & =  \secpar^{\alpha y} \cdot \left(\frac{1}{w}\right)^{\epsilon y}                               \\
		 & \leq  \secpar^{\alpha y} \cdot \left(\secpar^{-\frac{\alpha + c}{\epsilon}}\right)^{\epsilon y} \\
		 & =  \secpar^{-yc} \leq \secpar^{-c} = \negl
	\end{align*}
\end{proof}


We can now state and prove our main theorem.
\begin{theorem}
	\label{thm:gs-mdss-correctness}
	For all sets of parameters satisfying $\mdssRecThr > \sqrt{\mdssMaxShares \cdot \mdssPrivThr}$ and finite fields $\FF$ such that $|\FF| = \secpar^{\omega(1)}$ and $\mdssPrivThr + 1\leq (1 - \epsilon)\mdssRecThr$ for some constant $\epsilon$, the construction shown in \cref{fig:gs-mdss} satisfies $\mdssCVar$-correctness for a negligibly small $\mdssCVar(\secpar)$.
\end{theorem}

\newcommand{\mdssCorrectSet}{\set{\mdssSecret_i \text{ s.t. } |\indexSet_i| \geq \mdssRecThr}}
\newcommand{\mdssShortRecon}{\mathsf{R}}
\newcommand{\mdssCS}{C}

\begin{proof}
	In order to keep notation compact let $C = \mdssCorrectSet$ and $\mdssShortRecon = \mdssReconstructAlg$.
	\begin{align}
		     & \probsublong{\mdssShareList \gets \mdssGenerate\left(\set{\mdssConfig{i}}_{i\in[\mdssNumShareSets]}\right)}{\mdssShortRecon(\mdssShareList) \neq \mdssCS }                                                                                           \\
		=    & \probsublong{\mdssShareList \gets \mdssGenerate\left(\set{\mdssConfig{i}}_{i\in[\mdssNumShareSets]}\right)}{
			       \begin{gathered}
				       \exists \mdssSecret^* \text{ s.t. } \mdssSecret^* \in \mdssShortRecon(\mdssShareList) \wedge \mdssSecret^* \notin \mdssCS \\
				       \vee \\
				       \exists \mdssSecret^* \in \mdssCS \text{ s.t. } \mdssSecret^* \notin \mdssShortRecon(\mdssShareList)
			       \end{gathered}}                                                                                                                          \\
		\leq &
		\probsublong{\mdssShareList \gets \mdssGenerate\left(\set{\mdssConfig{i}}_{i\in[\mdssNumShareSets]}\right)}{
			\exists \mdssSecret^* \text{ s.t. } \mdssSecret^* \in \mdssShortRecon(\mdssShareList) \wedge \mdssSecret^* \notin \mdssCS}                                                                                                \notag                           \\
		     & + \probsublong{\mdssShareList \gets \mdssGenerate\left(\set{\mdssConfig{i}}_{i\in[\mdssNumShareSets]}\right)}{\exists \mdssSecret^* \in \mdssCS \text{ s.t. } \mdssSecret^* \notin \mdssShortRecon(\mdssShareList)}        \label{eq:gs-corr-union}  \\
		=    & \probsublong{\mdssShareList \gets \mdssGenerate\left(\set{\mdssConfig{i}}_{i\in[\mdssNumShareSets]}\right)}{
			       \exists \mdssSecret^* \text{ s.t. } \mdssSecret^* \in \mdssShortRecon(\mdssShareList) \wedge \mdssSecret^* \notin \mdssCS}                                                                                                \label{eq:gs-corr-GS}     \\
		=    & \probsublong{(\mdssPolySet, \mdssPointSet) \gets \mdssHonestDist{\FF}{\mdssPrivThr}{\set{(\mdssSecret_i, \mdssNumShares_i)}_{i \in [\mdssNumShareSets]}}}{
			       \exists \mdssSecret^* \text{ s.t. } \mdssSecret^* \in \mdssShortRecon(\mdssPointSet) \wedge \mdssSecret^* \notin \mdssCS}                                                                 \label{eq:gs-corr-const}                                  \\
		=    & \probsublong{(\mdssPolySet, \mdssPointSet) \gets \mdssHonestDist{\FF}{\mdssPrivThr}{\set{(\mdssSecret_i, \mdssNumShares_i)}_{i \in [\mdssNumShareSets]}}}{\bad_{\mdssPolySet, \mdssPointSet, \mdssPrivThr, \mdssRecThr}}                             \\
		\leq & \binom{|\mdssPointSet|}{\mdssRecThr} \cdot \mdssFieldBound                                                                                                                                                                  \label{eq:gs-corr-claim} \\
		\leq & \binom{\mdssMaxShares}{\mdssRecThr} \cdot \mdssFieldBound                                                                                                                                                                                            \\
		\leq & \negl\label{eq:gs-corr-asym}
	\end{align}

	Where \ref{eq:gs-corr-union} follows from the union bound, \ref{eq:gs-corr-GS} from \cref{thm:GS}, \ref{eq:gs-corr-const} from construction, \ref{eq:gs-corr-claim} from \cref{lem:mdss-corr-bound}, and \ref{eq:gs-corr-asym} from \cref{lem:mdss-asym-bound}.
\end{proof}
