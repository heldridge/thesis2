			\begin{algenumerate}
				\item Upon receiving $(\msg, \sid, \report_i)$ from a user $\user_i$, store the tuple $(\report_i)$ and send $(\msg, \sid, \user_i)$ to $\simulator$.
				\item Upon receiving $(\collect, \sid)$ from the server $\aggserver$, retrieve all tuples of the form $(\report_i)$ and return a random permutation of the reports $\{\report_1, \dots, \report_{\mdssWindowSize}\}$.
			\end{algenumerate}
