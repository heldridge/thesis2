We present here our main construction of an efficient PPB for threshold comparison.
Let $\pker$ and $\pkem$ be lossy public-key encryption schemes and define the following two relations:
\begin{align*}
	\kgrel^{} = & \left\{
	((pk', \com), (\thr, sk', r_k, r_c)) \::\:
	\begin{gathered}
		(sk', pk') = \kgen'(\lambda, \revpp, \matpp, \base, \numdigits, \thr; r_k) \\
		\wedge \: \com = \commit(\comparams, \thr; r_c)
	\end{gathered}
	\right\}              \\
	\erel^{} =  & \left\{
	((Z', \com, pk), (\uval, \msg, r_e, r_c)) \::\:
	\begin{gathered}
		Z' = \escrow'(\revpp, \matpp, \numdigits, pk, \uval, \msg; r_e) \\
		\wedge \: \com = \commit(\comparams, (\uval, \msg); r_c)
	\end{gathered}
	\right\}
\end{align*}

Let $\kglang^{}$ and $\elang^{}$ be the corresponding languages to these relations, and
$\kgps^{}$, $\eps^{}$ be corresponding NIZKs for these languages.
Our construction $(\maxt, \base)$-TCPPB can then be seen in \cref{fig:ppb-comp-const-01}, \cref{fig:ppb-comp-const-02}, and \cref{fig:ppb-comp-const-03}.
$\maxt$ and $\base$ are assumed to be implicit inputs to all algorithms, and $\msgspace_\rev$ and $\msgspace_\mat$ are the message spaces of $\pke_\rev$ and $\pke_\mat$, respectively, and are both assumed to be finite, cyclic, abelian groups with sizes $p_\rev$, $p_\mat$ and with generators $g_\rev$, $g_\mat$.

We can now state our main theorem.

\begin{theorem}
	\label{thm:ppbs-main}
	If $\pker$ is a weakly robust, lossy, \wlap{} public-key encryption scheme with message space $\msgspace_\rev$ and ciphertext space $\msgspace_\mat$, $\pkem$ is a lossy, $\mathcal{U}_{\msgspace_\mat}$-\slap{} public-key encryption scheme with message space $\msgspace_\mat$, $\msgspace_\rev$ and $\msgspace_\mat$ are both finite, cyclic, abelian groups, $(\pker, \pkem)$ satisfy lossy correlation resistance, $\comscheme = (\csetup, \commit)$ is a secure commitment scheme, $\kgps$ is zero-knowledge and strong simulation extractable, and $\eps$ is zero-knowledge and satisfies knowledge soundness, then \schemename{} is a secure $\fthr$-PPB scheme.
\end{theorem}

\subsection{Proofs}
As our construction uses the generic commit-and-prove technique from \cite{EC:KohLysNgu23}, the proofs of correctness of $\verescrow$ and $\verpk$, follow from the same arguments, and we omit the proofs here.

Before we prove the lemmas that will compose to our main theorem we first define some notation for iteratively applying the one-time-pad decryption operation.

Define $\rdec(x, y_1,\dots,y_n)$ as follows:
\procedureblock{$\rdec(x, y_1, \dots, y_n)$}{
	\otpk := x\\
	\pcfor i \in [n] \\
	\t \otpk := \otpk^{-1}\cdot y_i\\
	\pcreturn \otpk
}

\begin{lemma}
	\label{lemma:rdecchain}
	For a group with generator $g$, given $r_0, r_1, \dots, r_n$, one can efficiently compute a value $r'$ such that $\rdec(g^{r_0}, g^{r_1}, \dots, g^{r_n}) = (g^{r'})^{-1}\cdot g^{r_1}$.
\end{lemma}

\begin{proof}
	Let $f(r_0, r_2, \dots, r_n) = \sum_{i \in \{0, 2, \dots, n\}}((\prod_{n - i} -1)r_i) $

	We then have that:
	\begin{align*}
		\rdec(g^{r_0}, g^{r_1},\dots,g^{r_n})                                                & = \\
		(g^{r_0})^{(\prod_n-1)}\cdot (g^{r_1})^{(\prod_{n - 1}-1)} \cdot \dots \cdot g^{r_n} & = \\
		g^{f(r_0, r_2, \dots, r^n)} \cdot (g^{r_1})^{(\prod_{n - 1}-1)}
	\end{align*}

	Then, if $n$ is even we have that:
	\begin{align*}
		g^{f(r_0, r_2, \dots, r^n)} \cdot (g^{r_1})^{(\prod_{n - 1}-1)} & = g^{f(r_0, r_2, \dots, r^n)} \cdot g^{r_1}         \\
		                                                                & = (g^{-f(r_0, r_2, \dots, r^n)})^{-1} \cdot g^{r_1}
	\end{align*}
	and so $r' = -f(r_0, r_2, \dots, r^n)$.

	In the case that $n$ is odd we have that:
	\begin{align*}
		g^{f(r_0, r_2, \dots, r^n)} \cdot (g^{r_1})^{(\prod_{n - 1}-1)} & = g^{f(r_0, r_2, \dots, r^n)} \cdot g^{-r_1}                  \\
		                                                                & = g^{f(r_0, r_2, \dots, r^n)} \cdot g^{-r_1} g^{r_1} g^{-r_1} \\
		                                                                & = g^{f(r_0, r_2, \dots, r^n) - 2r_1} g^{r_1}                  \\
		                                                                & = (g^{-f(r_0, r_2, \dots, r^n) + 2r_1})^{-1} g^{r_1}          \\
	\end{align*}
	and so $r' = -f(r_0, r_2, \dots, r^n) + 2r_1$.
\end{proof}

We now begin proving our main theorem.

\subsubsection{Soundness}

\begin{lemma}
	\label{lemma:main-soundness}
	If $\Pi_{E}$ is a knowledge-sound NIZK, $\comscheme$ is a binding commitment scheme, $\pker$ satisfies weak-robustness, and $\pker$, $\pkem$ are correct and satisfy lossy correlation resistance, then \schemename{} satisfies soundness.
\end{lemma}

\begin{proof}
	Suppose for the sake of contradiction that there exists an adversary $\adv$ such that $\advantage{\soundgame}{\mathsf{TCPPB}, \adv} = \nu(\secpar)$ is non-negligible.
	In the adversary's first stage it outputs $(\thr, \kgr)$, and the challenger sets $(sk, pk) \leftarrow \kgen(\ppbpp, \thr, \kgr)$.
	In its second stage, the adversary $\adv$ outputs $(\ecom, (\uval, \msg), \er, \esc = (\esc', \zkproof_{E}))$.
	We can divide our analysis into the following three cases depending on the properties of these outputs.
	\begin{enumerate}
		\item There exist some $\er'$ and $(\uval', \msg') \neq (\uval, \msg)$ such that $\commit((\uval, \msg), \er) = \commit((\uval', \msg'), \er')$ and $\esc = \escrow(\ppbpp, pk, (\uval', \msg'), \er')$.
		\item $Z' \neq \escrow'(\pkerpp, \pkempp, \numdigits, pk, \uval, \msg)$ and (1) does not hold.
		\item Neither (1) nor (2) hold (i.e. $\adv$'s output is well-formed).
	\end{enumerate}
	Let the probability of $\adv$'s outputs satisfying property $j$ be denoted by $\nu_j(\secpar)$.
	If $\nu(\lambda)$ is non-negligible, then at least one of $\nu_1(\secpar)$, $\nu_2(\secpar)$, $\nu_3(\secpar)$ must be non-negligible.

	${}$

	Suppose that $\nu_2(\secpar)$ is non-negligible.
	In this case $\adv$ produced a proof of a false statement and we can construct a reduction $\bdv$ to the knowledge-soundness of $\Pi_{E}$.
	$\bdv$ runs $\adv$, simulates the generation of $pk$, receives $(\ecom, (\uval, \msg), \er, \esc = (\esc', \zkproof_{E}))$, and outputs $((Z', \ecom, pk), \zkproof_{E})$.
	As the statement is false any extractor must fail, meaning $\prob{\bdv \text{ wins}}$ is non-negligible, thus contradicting our assumption that $\Pi_{E}$ satisfies knowledge soundness.
	Therefore, $\nu_2(\secpar)$ is negligible.

	As $\nu_2(\secpar)$ is negligible we can assume there exists an extractor $\Ext_{\adv}$ that given $\trans_{\adv}$ outputs $(\uval', \msg', r_e, \er)$ such that $\ecom = \commit(\compp, (\uval', \msg'); \er)$ and $\esc'\allowbreak =\allowbreak \escrow'(\allowbreak\pkerpp,\allowbreak \pkempp,\allowbreak \numdigits,\allowbreak pk,\allowbreak \uval',\allowbreak \msg'; r_e)$.

	${}$

	Suppose $\nu_1(\secpar)$ is non-negligible.
	In this case we can break the computational binding property of the commitment scheme by using a reduction that uses the extractor and outputs $(\msg, \er, \msg', r_c)$.
	Therefore, $\nu_1(\secpar)$ must be negligible.

	${}$

	Suppose $\nu_3(\secpar)$ is non-negligible.
	In this case we can break a property of $\pker$ or $\pkem$.
	In all cases, the reduction begins by using the extractor to learn $(\uval', \msg', r_e, \er)$.
	The reduction can then use $r_e$ and $r_k$ to infer the randomness used to generate each individual value in $\escrow'$.
	Let $\rer_i$ and $\mer_i$ be the randomness used at iteration $i$ of $\escrow'$ for $\pker.\enc$ and $\pkem.\enc$, respectively.

	Let $\nu_3^{>}(\lambda)$ denote the probability that when $\nu_3(\secpar)$ is true,  $\uval' > \thr$.
	Let  $\nu_3^{<}(\lambda)$ be the case that $\uval' < \thr$, and $\nu_3^{=}(\secpar)$ be the case that $\uval' = \thr$.
	If $\nu_3(\lambda)$ is non-negligible, then at least one of $\nu_3^{>}(\lambda)$, $\nu_3^{<}(\lambda)$, $\nu_3^{=}(\secpar)$ must be non-negligible.

	${}$

	Suppose $\nu_3^{>}(\lambda)$ is non-negligible.
	In this case, let $d$ be the index for which $\uval'[d] > \thr[d]$ and $\forall i < d,\: \uval'[i] = \thr[i]$.
	We can split this into further cases depending on the behavior of $\dec$.
	To do so, we define the following notation.

	\newcommand{\badprobmat}[1]{\ensuremath{\nu_{3, #1}^{>,\mat}(\secpar)}}
	\newcommand{\badprobotpM}[1]{\ensuremath{\nu_{3, #1}^{>, \mathsf{OM}}(\secpar)}}
	\newcommand{\badprobotpR}[1]{\ensuremath{\nu_{3, #1}^{>, \mathsf{OR}}(\secpar)}}
	\newcommand{\badprobrev}[2]{\ensuremath{\nu_{3, #1, #2}^{>, \mathsf{\rev}}(\secpar)}}
	\newcommand{\badprobrevF}{\ensuremath{\nu_{3, d, \uval'[d]}^{>, \mathsf{*}}(\secpar)}}
	\begin{itemize}
		\item Let \badprobmat{i} denote the probability that $\pkem.\dec(\pkempp, \matsk_i^{\thr[i] + 1}, \mct_i) \allowbreak\neq\allowbreak \otpk_{i - 1} \cdot \otpk_i$ for some $i \in [d - 1]$.
		\item Let \badprobotpM{i} denote the probability that $\tilde{\otpk}_i \neq \otpk_i$ for some $i \in [0, d - 1]$
		\item Let \badprobotpR{i} denote the probability that $\revct_i \neq \pker.\enc(\pkerpp, \revpk_i^{\uval'[i]}, \otpk^*; \rer_i)$ for some $i \in [d]$.
		\item Let \badprobrev{i}{j} denote the probability that $\pker.\dec(\pkerpp, \revsk_i^j, \revct_i) \neq \bot$ for some $i \in [d - 1]$ and $j \in [\thr[i] + 1, \base]$, or for $i = d$ and $j \in [\thr[i] + 1, \uval'[d] - 1]$.
		\item Let \badprobrevF{} denote the probability that $\pker.\dec(\pkerpp, \revsk_d^{\uval'[d]}, \revct_d) \neq \otpk^*$.
	\end{itemize}

	If $\nu_3^{>}(\lambda)$ is non-negligible then at least one of the above must be non-negligible, as we must have $\tilde{\otpk}^* = \pker.\dec(\pkerpp, \revsk_d^{\uval'[d]}, \revct_d) \neq \otpk^*$ for the output of $\dec$ to not equal $m$.

	Suppose that $\badprobmat{i} > 0$ for some $i \in [n - 1]$.
	Then, we can break the correctness property of $\pkem$, as
	$$
		\pkem.\dec(\pkempp, sk, \pkem.\enc(\pkempp, pk, \msg'; \mer_i)) \neq \msg'
	$$
	Therefore, $\badprobmat{i} = 0$ for all $i \in [d - 1]$.

	Next, it is clearly the case that $\badprobotpM{0} = 0$, as $\otpk_0'$ is set identically to how it is set in $\escrow'$.
	We now show that for all $i \in [n - 1]$, if $\badprobmat{i - 1} = 0$ then $\badprobmat{i} = 0$.
	As $\badprobmat{i - 1} = 0$, we have that $\otpct_i^{\mat} = \otpk_{i - 1}\cdot \otpk_i$.
	Then, by definition and our assumption we have that $\tilde{\otpk}_i = (\otpk_{i - 1})^{-1}\cdot \otpk_{i - 1}\cdot \otpk_i = \otpk_i$.
	Thus, by induction, we have that $\badprobotpM{i} = 0$ for all $i \in [0, d - 1]$.

	Next, consider the case that $\badprobotpR{i} > 0$ for some $i$.
	This is equivalent to stating that $(\tilde{\otpk}_{i - 1})^{-1}\cdot \otpct_i^{\rev} \neq \pker.\enc(\pkerpp, \revpk_i^{\uval'[i]}, \otpk^*; \rer_i)$.
	Then, by the definition of $\otpct^{\rev}_i$ and the fact that $\badprobotpM{i - 1} = 0$, we have:
	\begin{align*}
		\otpk_{i - 1}^{-1} \cdot  \otpk_{i - 1}\cdot \pker.\enc(\pkerpp, \revpk_i^{\uval'[i]}, \otpk^*; \rer_i) & \neq
		\pker.\enc(\pkerpp, \revpk_i^{\uval'[i]}, \otpk^*; \rer_i)                                                                                                                \\
		\pker.\enc(\pkerpp, \revpk_i^{\uval'[i]}, \otpk^*; \rer_i)                                              & \neq \pker.\enc(\pkerpp, \revpk_i^{\uval'[i]}, \otpk^*; \rer_i)
	\end{align*}
	We therefore have that $\badprobotpR{i} = 0$ for all $i \in [d]$.

	Next, suppose that $\badprobrev{i}{j}$ is non-negligible for some $(i, j)$.
	Then, we have that:
	\begin{align*}
		\pker.\dec(\pkerpp, \revsk_i^j, \revct_i) \neq \bot                                                                              \\
		\pker.\dec(\pkerpp, \revsk_i^j, \pker.\enc(\pkerpp, \revpk_i^{\uval'[i]}, \otpk^*)) \neq \bot \tag{\badprobotpR{i} = 0}          \\
		\pker.\dec(\pkerpp, \revsk_i^j, \pker.\enc(\pkerpp, \revpk_i^{\uval'[i]}, \otpk^*; \rer_i)) \neq \bot \tag{Definition of $\enc$} \\
	\end{align*}

	Then, we can write a reduction $\bdv$ that breaks the weak-robustness property of $\pker$.
	The reduction outputs $(0, 1, \otpk^*, \rer_i)$ and wins as $\uval'[i] < j$.
	Therefore, $\badprobrev{i}{j}$ must be negligible for all $i \in [d - 1]$ and $j \in [t[i] + 1, \base]$ or when $i = d$ and $j \in [\thr[i] + 1, \uval'[d] - 1]$.

	Finally, suppose that $\badprobrevF{} > 0$.
	Then, we have that:
	\begin{align*}
		\pker.\dec(\pkerpp, \revsk_i^{\uval'[i]}, \revct_i) \neq \otpk^*                                                                     \\
		\pker.\dec(\pkerpp, \revsk_i^{\uval'[i]}, \pker.\enc(\pkerpp, \revpk_i^{\uval'[i]}, \otpk^*)) \neq \otpk^* \tag{\badprobotpR{i} = 0} \\
	\end{align*}
	This breaks the correctness of $\pker$, and therefore we have that $\badprobrevF{} = 0$, and that
	$\nu_3^{>}(\lambda)$ is negligible.

	\newcommand{\badprobrevnonbot}[2]{\ensuremath{\nu_{3, #1, #2}^{<, \mathsf{\rev}}}}
	Suppose $\nu_3^{<}(\lambda)$ is non-negligible.
	In this case, let $d$ be the index for which $\uval'[d] < \thr[d]$ and $\forall i < d, \: \uval'[i] = \thr[i]$.
	Let $\badprobrevnonbot{i}{j}(\secpar)$ denote the probability that $\pker.\dec(\pkerpp, \revsk_i^j, \revct_i) \allowbreak\neq\allowbreak \bot$ for some $i \in [\numdigits]$ and $j \in [\thr[i] + 1, \base]$.
	If $\nu_3^{<}(\secpar)$ is non-negligible, then at least one of $\badprobrevnonbot{i}{j}(\secpar)$ must be non-negligible.
	By the same logic as above, we have that $\badprobrevnonbot{i}{j}(\secpar) = 0$ for all $i \in [d]$.

	Now, suppose that $\badprobrevnonbot{i}{j}(\secpar)$ is non-negligible for some $i > d$.
	By definition we have that there is a non-negligible probability that $\pker.\dec(\revsk_i^j, \revct_i) \neq \bot$.

	We then have:
	\begin{align*}
		\pker.\dec(\revsk_i^j, \revct_i)                                       & \neq \bot \\
		\pker.\dec(\revsk_i^j,\tilde{\otpk}_{i-1}^{-1} \cdot \otpct_i^{\rev})) & \neq \bot \\
	\end{align*}
	\vspace{-6em}
	\begin{align*}
		 & \pker.\dec\left(\revsk_i^j, \left(\rdec(\otpk_{d - 1}, \pkem.\dec(\matsk_d^{\thr[d] + 1}, \matct_d), \otpct_{d}^\mat, \dots, \otpct_{i - 1}^\mat)
		\right)^{-1} \cdot \otpct_i^{\rev} \right)                                                                                                           \\
		 & \neq \bot
	\end{align*}
	% \vspace{-5em}
	\begin{align*}
		\pker.\dec(\revsk_i^j, (g_C^{-r'}\cdot \pkem.\dec(\matsk_d^{\thr[d] + 1}, \matct_d))^{-1}\cdot \otpct_i^{\rev}) & \neq \bot \tag{\cref{lemma:rdecchain}}
	\end{align*}

	As $\matct_d$ was encrypted under a lossy key (as $\uval[d] < \thr[d]$), we can then break the lossy-correlation-resistance of $(\pker, \pkem)$ with a reduction that outputs
	$$
		(\mer_d,  \rer_i, r', \okgr_{i - 1}, \okgr_{d - 1} + \okgr_d, \okgr_*)
	$$
	Therefore, $\badprobrevnonbot{i}{j}(\secpar) \leq \negl$ for all $i \in [d + 1, \numdigits]$
	and for all $j \in [\thr[i] + 1, \base]$, and so $\nu_3^{<}(\lambda)$ must be negligible.

	Finally, we have already shown that $\badprobrev{i}{j}$ must be negligible at all positions where $\uval'[i] = \thr[i]$.
	Therefore, $\nu_3^{=}(\secpar)$ must be negligible.
	In turn, $\nu_3(\secpar)$ and then $\nu(\secpar)$ must be negligible, and therefore the scheme satisfies soundness.
\end{proof}

\subsubsection{Correctness}
\begin{lemma}
	\label{lemma:main-correctness}
	If $\Pi_{KG}$ and $\Pi_{E}$ are complete and $\pker$ and $\pkem$ are correct, then \schemename{} satisfies correctness of $\dec$.
\end{lemma}

\begin{proof}

	We first consider the case that $\uval > \thr$. In this case, there is some index $d$ for which $\uval[d] > \thr[d]$, and $\uval[i] = \thr[i]$ $ $ $\forall i < d$.
	For $i < d$, as $\uval[i] = \thr[i]$, the match decryption $\otpct_i^{\mat} \leftarrow \pkem.\dec(\cdot)$ succeeds (as $\matpk_i^{v[i] + 1}$ must be injective), which in turn implies that  $\tilde{\otpk}_i$ is computed correctly.
	Therefore, the ciphertext $\ct_d^{\rev}$ is properly formed.
	Finally, as $\uval[d] > \thr[d]$, the auditor will attempt to decrypt $\ct_d^{\rev}$ with $\revsk_d^{\uval[d]}$, which will correctly return $\otpk^*$ and then $\msg$.

	Next we turn to the case that $\uval < \thr$. In this case there is some index $d$ for which $\uval[d] < \thr[d]$, and $\uval[i] = \thr[i]$ $ $ $\forall i < d$.
	As $\uval[i] = \thr[i]$ for all $i < d$, $\pker.\dec(\cdot)$ will output $\bot$ for each tried $\revsk_i^j$ by the weak robustness property.
	Finally, at index $d$ the match decryption will \emph{fail}, resulting in all future reveal ciphertexts being blinded, and therefore $\pker.\dec(\cdot)$ outputting $\bot$ by the lossy correlation resistance of $(\pker, \pkem)$ (see the proof of \cref{lemma:main-soundness} for the concrete reduction).
	As all reveal decrypt attempts fail, $\dec$ will therefore output $\bot$ as required.

	Finally, we examine the case that $\uval = \thr$.
	In this case all reveal ciphertexts are encrypted under lossy keys, and therefore $\dec$ will always output $\bot$ by the weak robustness property of $\pker$.
\end{proof}

\subsubsection{Threshold Hiding}

\begin{lemma}
	\label{lemma:main-th}
	If $\pker$ and $\pkem$ are lossy encryption schemes satisfing indistinguishability of keys and $\Pi_{KG}$ satisfies zero-knowledge, then \schemename{} satisfies threshold hiding.
\end{lemma}

\begin{proof}
	We prove security using a sequence of hybrids.

	\begin{itemize}

		\item $\newhyb$: This hybrid is identical to $\thrhidegame_0$.

		\item $\nexthyb$: In this hybrid the proof $\zkproof_{KG}$ output by $\kgen(\cdot)$ is generated by the $\Pi_{KG}$ simulator $\Sim$. By the zero-knowledge property of $\Pi_{KG}$, $\prevhyb \pindist \currhyb$.

		\item $\nexthyb$: In this hybrid steps $1$ through $10$ of $\kgen'$ are replaced with the following:
		      \begin{pchstack}[center, space=1em]
			      \procedure[linenumbering]{}{
				      \pcfor i \in [\numdigits]\\
				      \pcind \pcfor j \in [0, b - 1] \\
				      \t[2] \pcif j \leq t_0[i] \\
				      \t[3] \revpk_i^j \leftarrow \pker.\kgenlossy(\revpp, \secparam) \\
				      \t[2] \pcelse \\
				      \t[3] \revsk_i^j, \revpk_i^j \leftarrow \pker.\kgen(\revpp, \secparam) \\
				      \t[2] \pcif j \leq t_0[i] \\
				      \t[3] \matpk_i^j \leftarrow \pkem.\kgenlossy(\matpp, \secparam) \\
				      \t[2] \pcelse \\
				      \t[3] \matsk_i^j, \matpk_i^j \leftarrow \pkem.\kgen(\matpp, \secparam) \\
				      \t[2] \pairpk_i^j \leftarrow (\revpk_i^j, \matpk_i^j) \\
				      \t[2] \pairsk_i^j \leftarrow \bot
			      }
		      \end{pchstack}
		      As $\adv$ does not have access to $sk$, this hybrid behaves identically to $\prevhyb$ from the adversary's point of view, and so $\prevhyb \pindist \currhyb$.

		\item $\nexthyb$: In this hybrid step $3$ of $\kgen'$ is changed from $j \leq t_0[i]$ to $j \leq t_1[i]$.
		      We show that $\prevhyb \cindist \currhyb$ via a series of sub-hybrids.
		      In hybrid $\hyb'_{k, \ell}$ for $k \in [0, \numdigits]$ and $\ell \in [0, \base - 1]$, step $3$ is changed as described at loop iteration $(k, \ell)$.
		      Clearly $\hyb'_{0, 0}$ is equivalent to $\prevhyb$, and $\hyb'_{\numdigits, \base - 1}$ is equivalent to $\currhyb$.

		      In the case that $(\ell \leq t_0[k]) = (\ell \leq t_1[k])$, the two hybrids are identical.
		      In the case that $(\ell \leq t_0[k]) \neq (\ell \leq t_1[k])$, indistinguishability follows from the key indistinguishability property of $\pker$.
		      We therefore have that $\hyb'_{k, \ell} \cindist \hyb'_{k, \ell + 1}$, and $\hyb'_{k, \base} \cindist \hyb'_{k + 1, 0}$ for all $k, \ell$, and therefore that $\prevhyb \cindist \currhyb$.

		\item $\nexthyb$: In this hybrid step $7$ of $\kgen'$ is changed from $j \leq t_0[i]$ to $j \leq t_1[i]$.
		      $\prevhyb \cindist \currhyb$ follows identically as before from the key indistinguishability property of $\pkem$.

		\item $\nexthyb$ In this hybrid $\kgen'$ is returned to its original formulation, with all comparisons performed against $t_1$.
		      As $\adv$ does not have access to $sk$, this hybrid behaves identically to $\prevhyb$, and so $\prevhyb \pindist \currhyb$.

		\item $\nexthyb$: This hybrid is identical to $\thrhidegame_1$.
		      By the zero-knowledge property of $\Pi_{KG}$ we have that $\prevhyb \pindist \currhyb$.
	\end{itemize}


	We therefore conclude that $\hyb_0 \cindist \hyb_6$, and thus \schemename{} satisfies threshold hiding.
\end{proof}
